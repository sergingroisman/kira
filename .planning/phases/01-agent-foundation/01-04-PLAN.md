import React, { useState, useEffect } from "react";
import { Text } from "ink";

const FRAMES = ["⠋", "⠙", "⠹", "⠸", "⠼", "⠴", "⠦", "⠧", "⠇", "⠏"];

export function Spinner({ label = "Thinking..." }: { label?: string }) {
  const [frame, setFrame] = useState(0);

  useEffect(() => {
    const id = setInterval(() => setFrame(f => (f + 1) % FRAMES.length), 80);
    return () => clearInterval(id);
  }, []);

  return <Text color="cyan" dimColor>{FRAMES[frame]} {label}</Text>;
}

1. **Token batching** (Pitfall 4 guard — 16ms interval):
```typescript
const tokenBuffer = useRef("");
const flushTimer = useRef<ReturnType<typeof setInterval> | null>(null);

const startFlush = useCallback(() => {
  if (flushTimer.current) return;
  flushTimer.current = setInterval(() => {
    if (tokenBuffer.current) {
      setStreaming(prev => prev + tokenBuffer.current);
      tokenBuffer.current = "";
    }
  }, 16);
}, []);

const stopFlush = useCallback(() => {
  if (flushTimer.current) { clearInterval(flushTimer.current); flushTimer.current = null; }
  if (tokenBuffer.current) {
    setStreaming(prev => prev + tokenBuffer.current);
    tokenBuffer.current = "";
  }
}, []);
```
