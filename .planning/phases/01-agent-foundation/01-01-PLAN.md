```typescript
import { ChatDeepSeek } from "@langchain/deepseek";
import type { GraphState } from "../state.js";

// LLM is initialized once at module load and reused across all calls.
// Tool binding happens in graph.ts after ToolRegistry is initialized.
// In the Walking Skeleton, no tools are bound — plain invoke.
let _llm: ReturnType<typeof buildLLM> | null = null;

function buildLLM() {
  return new ChatDeepSeek({
    apiKey: process.env.DEEPSEEK_API_KEY,
    model: "deepseek-chat",   // NOT deepseek-reasoner; reasoner has no tool calling
    temperature: 0,
    maxRetries: 2,
  });
}

// getLLM returns a memoized instance.
// graph.ts replaces this with a tool-bound LLM in Plan 02.
export function getLLM() {
  if (!_llm) _llm = buildLLM();
  return _llm;
}

// bindLLMTools replaces the LLM singleton with a tool-bound version.
// Called once from graph.ts after ToolRegistry is populated (Plan 02).
export function bindLLMTools(tools: Parameters<ReturnType<typeof buildLLM>["bindTools"]>[0]) {
  _llm = buildLLM().bindTools(tools, { tool_choice: "auto" });
}

export async function callLLM(state: GraphState): Promise<Partial<GraphState>> {
  const llm = getLLM();
  const response = await llm.invoke(state.messages);
  // MessagesAnnotation reducer appends — return array with single new message
  return { messages: [response] };
}
```

**agent/src/ui/InputBar.tsx**:
```tsx
import React, { useState } from "react";
import { Box, Text, useInput } from "ink";

interface InputBarProps {
  onSubmit: (value: string) => void;
  disabled?: boolean;
}

export function InputBar({ onSubmit, disabled = false }: InputBarProps) {
  const [buffer, setBuffer] = useState("");

  // useInput from ink uses useLayoutEffect internally — raw mode is set
  // synchronously during commit phase. Do NOT call process.stdin.setRawMode() directly.
  useInput(
    (input, key) => {
      if (key.return) {
        const trimmed = buffer.trim();
        if (trimmed) {
          onSubmit(trimmed);
          setBuffer("");
        }
        return;
      }
      if (key.backspace || key.delete) {
        setBuffer(prev => prev.slice(0, -1));
        return;
      }
      if (key.escape) {
        setBuffer("");
        return;
      }
      // Ignore control characters (Ctrl+C handled by exitOnCtrlC in render())
      if (!key.ctrl && !key.meta && input) {
        setBuffer(prev => prev + input);
      }
    },
    { isActive: !disabled }
  );

  return (
    <Box>
      <Text color="green">{">"} </Text>
      <Text>{buffer}</Text>
      <Text color="gray">{"█"}</Text>
    </Box>
  );
}
```

**agent/src/ui/App.tsx**:
```tsx
import React, { useState, useCallback, useRef } from "react";
import { Box, Text, Static } from "ink";
import { HumanMessage } from "@langchain/core/messages";
import { graph, makeSessionConfig } from "../graph/graph.js";
import { InputBar } from "./InputBar.js";
import { StreamingOutput } from "./StreamingOutput.js";
import { Spinner } from "./Spinner.js";

interface Message {
  role: "user" | "assistant";
  content: string;
}

interface AppProps {
  sessionId?: string;
}

export function App({ sessionId }: AppProps) {
  const [history, setHistory] = useState<Message[]>([]);
  const [streaming, setStreaming] = useState("");
  const [isThinking, setIsThinking] = useState(false);

  // Session config is stable for the lifetime of the process (one thread_id per session — D-06)
  const sessionConfig = useRef(makeSessionConfig());

  // Token accumulator ref — avoids excessive setState on rapid token arrival (Pitfall 4)
  const tokenBuffer = useRef("");
  const flushTimer = useRef<ReturnType<typeof setInterval> | null>(null);

  const startFlush = useCallback(() => {
    if (flushTimer.current) return;
    flushTimer.current = setInterval(() => {
      if (tokenBuffer.current) {
        setStreaming(prev => prev + tokenBuffer.current);
        tokenBuffer.current = "";
      }
    }, 16); // ~60fps flush rate
  }, []);

  const stopFlush = useCallback(() => {
    if (flushTimer.current) {
      clearInterval(flushTimer.current);
      flushTimer.current = null;
    }
    // Flush any remaining tokens
    if (tokenBuffer.current) {
      setStreaming(prev => prev + tokenBuffer.current);
      tokenBuffer.current = "";
    }
  }, []);

  const handleSubmit = useCallback(async (userInput: string) => {
    setIsThinking(true);
    setStreaming("");
    tokenBuffer.current = "";

    // Add user message to visible history immediately
    setHistory(prev => [...prev, { role: "user", content: userInput }]);

    startFlush();

    let fullResponse = "";

    try {
      // graph.stream with streamMode: "messages" yields [messageChunk, metadata] tuples (Pattern 5)
      // LangGraph intercepts the LLM call at framework level — callLLM uses llm.invoke() (Pitfall 2)
      for await (const [chunk] of await graph.stream(
        { messages: [new HumanMessage(userInput)] },
        { ...sessionConfig.current, streamMode: "messages" }
      )) {
        const text = typeof chunk.content === "string" ? chunk.content : "";
        if (text) {
          tokenBuffer.current += text;
          fullResponse += text;
        }
      }
    } catch (err: unknown) {
      const message = err instanceof Error ? err.message : String(err);
      fullResponse = `Error: ${message}`;
      tokenBuffer.current = fullResponse;
    } finally {
      stopFlush();
      setIsThinking(false);
      setHistory(prev => [...prev, { role: "assistant", content: fullResponse }]);
      setStreaming("");
    }
  }, [startFlush, stopFlush]);

  return (
    <Box flexDirection="column" paddingX={1}>
      {/* Static renders completed history without re-rendering previous items */}
      <Static items={history}>
        {(msg, i) => (
          <Box key={i} flexDirection="column" marginBottom={1}>
            <Text color={msg.role === "user" ? "green" : "white"} bold={msg.role === "user"}>
              {msg.role === "user" ? "You" : "Sergin"}
            </Text>
            <Text>{msg.content}</Text>
          </Box>
        )}
      </Static>

      {isThinking && <Spinner />}
      {streaming && <StreamingOutput tokens={streaming} />}
```
