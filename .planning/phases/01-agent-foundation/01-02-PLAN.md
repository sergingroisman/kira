```typescript
import type { tool } from "@langchain/core/tools";

// ToolDefinition is the plugin contract.
// Phase 2 will add: loadFromDir(dir: string): Promise<void> to the class.
// Phase 3 (MCP) will register MCP-sourced tools via the same register() method.
export interface ToolDefinition {
  name: string;
  lcTool: ReturnType<typeof tool>;
}

export class ToolRegistry {
  private readonly tools: ToolDefinition[] = [];

  register(def: ToolDefinition): void {
    if (this.tools.some(t => t.name === def.name)) {
      throw new Error(`Tool "${def.name}" is already registered`);
    }
    this.tools.push(def);
  }

  getLCTools(): ReturnType<typeof tool>[] {
    return this.tools.map(d => d.lcTool);
  }

  getNames(): string[] {
    return this.tools.map(d => d.name);
  }
}

// Singleton registry — initialized at startup, stable reference for lifetime of process
export const registry = new ToolRegistry();
```

**agent/src/tools/bash.ts**:
```typescript
import { tool } from "@langchain/core/tools";
import { exec } from "child_process";
import { promisify } from "util";
import { z } from "zod";
import type { ToolDefinition } from "./registry.js";

const execAsync = promisify(exec);

// Bash tool — unrestricted per D-09.
// Single-user system; owner controls the machine. No timeout, no blocklist.
// Returns stdout on success, error string on failure (does NOT throw).
// Error return means ToolNode emits a ToolMessage with that error string,
// which the reflect node then injects back to the LLM state.
const bashLcTool = tool(
  async ({ command }: { command: string }) => {
    try {
      const { stdout, stderr } = await execAsync(command, {
        shell: "/bin/zsh",
        maxBuffer: 10 * 1024 * 1024, // 10MB output cap
      });
      return stdout || stderr || "(no output)";
    } catch (err: unknown) {
      const e = err as { message: string; stderr?: string };
      return `Error: ${e.message}\n${e.stderr ?? ""}`.trim();
    }
  },
  {
    name: "Bash",
    description:
      "Execute a shell command in zsh and return its output. Use for: file listing, running scripts, git operations, system info, package management. Returns stdout on success or error details on failure.",
    schema: z.object({
      command: z.string().describe("The shell command to execute"),
    }),
  }
);

export const bashTool: ToolDefinition = {
  name: "Bash",
  lcTool: bashLcTool,
};
```

```typescript
import { tool } from "@langchain/core/tools";
import { writeFile, appendFile, mkdir } from "fs/promises";
import { dirname } from "path";
import { z } from "zod";
import type { ToolDefinition } from "./registry.js";

const fileWriteLcTool = tool(
  async ({ path, content }: { path: string; content: string }) => {
    try {
      // Create parent directories if they don't exist
      await mkdir(dirname(path), { recursive: true });
      await writeFile(path, content, "utf-8");
      return `Written ${Buffer.byteLength(content, "utf-8")} bytes to ${path}`;
    } catch (err: unknown) {
      const e = err as { message: string };
      return `Error: ${e.message}`;
    }
  },
  {
    name: "FileWrite",
    description:
      "Write content to a file, creating it if it does not exist and overwriting if it does. Creates parent directories automatically.",
    schema: z.object({
      path: z.string().describe("Absolute or relative path to the file to write"),
      content: z.string().describe("Content to write to the file"),
    }),
  }
);

const fileAppendLcTool = tool(
  async ({ path, content }: { path: string; content: string }) => {
    try {
      await mkdir(dirname(path), { recursive: true });
      await appendFile(path, content, "utf-8");
      return `Appended ${Buffer.byteLength(content, "utf-8")} bytes to ${path}`;
    } catch (err: unknown) {
      const e = err as { message: string };
      return `Error: ${e.message}`;
    }
  },
  {
    name: "FileAppend",
    description:
      "Append content to an existing file. Creates the file if it does not exist.",
    schema: z.object({
      path: z.string().describe("Absolute or relative path to the file"),
      content: z.string().describe("Content to append to the file"),
    }),
  }
);

export const fileWriteTool: ToolDefinition = {
  name: "FileWrite",
  lcTool: fileWriteLcTool,
```

**agent/src/graph/nodes/reflect.ts**:
```typescript
import { ToolMessage } from "@langchain/core/messages";
import type { GraphState } from "../state.js";

// Reflect node (D-05): examines the last ToolMessage in state.
// If it contains an error, it enriches the message so the LLM has a clear signal.
// The LLM then decides: retry with corrected parameters, or explain the failure to the user.
// No fixed retry counter for Phase 1 — LLM decides when to stop.
//
// Edge: executeTool → reflect → callLLM (always loops back — the LLM decides to stop, not the graph)
export async function reflect(state: GraphState): Promise<Partial<GraphState>> {
  const messages = state.messages;
  const last = messages[messages.length - 1];

  // Only enrich ToolMessages that look like errors
  if (
    last instanceof ToolMessage &&
    typeof last.content === "string" &&
    last.content.startsWith("Error:")
  ) {
    const enriched = new ToolMessage({
      tool_call_id: last.tool_call_id,
      content:
        `${last.content}\n\n` +
        `[Reflect] The tool returned an error. ` +
        `Please either: (a) retry the tool call with corrected parameters, ` +
        `or (b) explain to the user why this cannot be completed.`,
    });
    return { messages: [enriched] };
  }

  // Happy path: no error — pass through without adding messages
  return { messages: [] };
}
```

```typescript
import { makeToolNode } from "./nodes/executeTool.js";
import { reflect } from "./nodes/reflect.js";

import { bashTool } from "../tools/bash.js";
import { fileReadTool, } from "../tools/fileRead.js";
import { fileWriteTool, fileAppendTool } from "../tools/fileWrite.js";

// ── Tool Registration ─────────────────────────────────────────────────────────
// Phase 1 tools: Bash, FileRead, FileWrite, FileAppend
// Phase 2 will add: registry.loadFromDir('./skills/')
// Phase 3 will add: registry.register(mcpTool) for each MCP-sourced tool

registry.register(fileReadTool);
registry.register(fileWriteTool);
registry.register(fileAppendTool);

// ── LLM Binding ──────────────────────────────────────────────────────────────

// ── Graph Construction ────────────────────────────────────────────────────────
const memory = new MemorySaver();
const toolNode = makeToolNode(registry);

// Full Phase 1 topology (D-04):
//   START → callLLM → routeNext ──(no tools)──→ END
//                         │
//                    (has tool_calls)

export const graph = new StateGraph(MessagesAnnotation)
  .addNode("callLLM", callLLM)
  .addNode("executeTool", toolNode)
  .addNode("reflect", reflect)
  .addEdge(START, "callLLM")
  .addConditionalEdges("callLLM", routeNext)
  .addEdge("executeTool", "reflect")
  .addEdge("reflect", "callLLM")
  .compile({ checkpointer: memory });

// Session config factory — call once per CLI session, reuse on every turn (Pitfall 3 guard)
export function makeSessionConfig() {
  return { configurable: { thread_id: crypto.randomUUID() } };
}
```

```tsx
  toolName: string;
  args?: string;
  result?: string;
  status: "running" | "done" | "error";
}

const MAX_PREVIEW = 120;

function truncate(str: string): string {
  return str.length > MAX_PREVIEW ? str.slice(0, MAX_PREVIEW) + "…" : str;
}

export function ToolCallPanel({ toolName, args, result, status }: ToolCallPanelProps) {
  const statusColor = status === "running" ? "yellow" : status === "error" ? "red" : "green";
  const statusSymbol = status === "running" ? "⟳" : status === "error" ? "✗" : "✓";

  return (
    <Box
      flexDirection="column"
      borderStyle="single"
      borderColor={statusColor}
      paddingX={1}
      marginY={0}
    >
      <Box>
        <Text color={statusColor}>{statusSymbol} </Text>
        <Text bold color="cyan">{toolName}</Text>
        {args && <Text color="gray"> {truncate(args)}</Text>}
      </Box>
      {result && (
        <Text color={status === "error" ? "red" : "white"}>
          {truncate(result)}
        </Text>
      )}
    </Box>
  );
}
```
